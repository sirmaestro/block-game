cmake_minimum_required(VERSION 3.24)

# std::format in Apple's libc++ needs macOS 13.3
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.3" CACHE STRING "Minimum macOS version")

project(
    block
    VERSION "1.5.0"
    DESCRIPTION "Source rebuild of BLOCK (KENTA, 2000) on SDL3"
    LANGUAGES C CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# static C/C++ runtime, so block.exe runs without the Visual C++ redistributable
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(
    SDL3
    GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
    GIT_TAG release-3.2.8
    GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(SDL3)
if(APPLE)
    # SDL keeps calling older Cocoa/Metal APIs for compatibility; with our 13.3 deployment
    # target those show up as deprecation warnings in SDL's own sources
    target_compile_options(SDL3-static PRIVATE -Wno-deprecated-declarations)
endif()

# compile assets/ into the executable, so it is a single file
set(embedded_sources)
function(embed_asset input symbol)
    set(output ${CMAKE_BINARY_DIR}/embedded/${symbol}.cpp)
    add_custom_command(
        OUTPUT ${output}
        COMMAND ${CMAKE_COMMAND} -DINPUT=${input} -DOUTPUT=${output} -DNAME=${symbol}
                -P ${CMAKE_SOURCE_DIR}/cmake/embed.cmake
        DEPENDS ${input} ${CMAKE_SOURCE_DIR}/cmake/embed.cmake
        COMMENT "Embedding ${input}")
    set(embedded_sources ${embedded_sources} ${output} PARENT_SCOPE)
endfunction()
foreach(asset IN ITEMS sprites.bmp stages.txt icon.bmp)
    string(MAKE_C_IDENTIFIER "${asset}" symbol)
    embed_asset(${CMAKE_SOURCE_DIR}/assets/${asset} ${symbol})
endforeach()

# the sounds, plus a lookup by name (assets::sound)
file(GLOB sound_files CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/assets/sounds/*.ogg)
set(sound_declarations "")
set(sound_lookup "")
foreach(sound IN LISTS sound_files)
    get_filename_component(stem ${sound} NAME_WE)
    string(MAKE_C_IDENTIFIER "sound_${stem}" symbol)
    embed_asset(${sound} ${symbol})
    string(APPEND sound_declarations "std::span<const std::uint8_t> ${symbol}();\n")
    string(APPEND sound_lookup "    if (name == \"${stem}\")\n    {\n        return ${symbol}();\n    }\n")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_BINARY_DIR}/embedded/sound_table.cpp CONTENT "// generated by CMakeLists.txt - do not edit
#include <cstdint>
#include <span>
#include <string_view>

namespace assets
{

${sound_declarations}
std::span<const std::uint8_t> sound(std::string_view name)
{
${sound_lookup}    return {};
}

}
")
list(APPEND embedded_sources ${CMAKE_BINARY_DIR}/embedded/sound_table.cpp)

# Ogg Vorbis decoder (public domain), built without the project's warning flags
add_library(stb_vorbis STATIC third_party/stb/stb_vorbis_impl.c)
target_include_directories(stb_vorbis PUBLIC third_party/stb)
target_compile_definitions(stb_vorbis PUBLIC STB_VORBIS_NO_STDIO STB_VORBIS_NO_PUSHDATA_API)
if(NOT MSVC)
    target_compile_options(stb_vorbis PRIVATE -w)
endif()

# the game itself, shared by the executable and the tests
add_library(block_core STATIC
    src/audio.cpp
    src/dmath.cpp
    src/game.cpp
    src/gfx.cpp
    src/physics.cpp
    src/replay.cpp
    src/save.cpp
    src/screens.cpp
    src/settings.cpp
    src/stages.cpp
    src/voxels.cpp
    src/windows.cpp
    ${embedded_sources}
)
target_include_directories(block_core PUBLIC src)
target_compile_features(block_core PUBLIC cxx_std_23)
target_link_libraries(block_core PUBLIC SDL3::SDL3-static stb_vorbis)
if(MSVC)
    target_compile_options(block_core PUBLIC /W4 /utf-8)
else()
    # no fused multiply-add: replays must give the same bits on every platform (see src/dmath.hpp)
    target_compile_options(block_core PUBLIC -Wall -Wextra -ffp-contract=off)
endif()

add_executable(block WIN32 MACOSX_BUNDLE src/main.cpp)
target_link_libraries(block PRIVATE block_core)

if(WIN32)
    target_sources(block PRIVATE src/block.rc)
endif()

if(APPLE)
    set_target_properties(block PROPERTIES
        OUTPUT_NAME BLOCK
        MACOSX_BUNDLE_BUNDLE_NAME BLOCK
        MACOSX_BUNDLE_GUI_IDENTIFIER jp.kenta.block
        MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
        MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR})
endif()

# release package: cmake --build build --config Release --target package
if(APPLE)
    install(TARGETS block BUNDLE DESTINATION .)
    set(package_platform macos)
else()
    install(TARGETS block RUNTIME DESTINATION .)
    if(WIN32)
        set(package_platform win64)
    else()
        set(package_platform linux-${CMAKE_SYSTEM_PROCESSOR})
    endif()
endif()
install(FILES README.md LICENSE DESTINATION .)
install(FILES assets/stages.txt DESTINATION . RENAME stage-format-example.txt)

option(BLOCK_TESTS "Build the tests" ON)
if(BLOCK_TESTS)
    enable_testing()
    add_executable(block_tests tests/tests.cpp)
    target_link_libraries(block_tests PRIVATE block_core)
    target_compile_definitions(block_tests PRIVATE BLOCK_TEST_DATA="${CMAKE_SOURCE_DIR}/tests/data")
    add_test(NAME block_tests COMMAND block_tests)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(CPACK_GENERATOR TGZ)
else()
    set(CPACK_GENERATOR ZIP)
endif()
set(CPACK_PACKAGE_FILE_NAME "BLOCK-${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}-${package_platform}")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
include(CPack)
